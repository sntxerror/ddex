# DDEX production nginx config for privatesecure.org
# - Serves the static PWA from /var/www/privatesecure.org
# - Proxies /test/* and /prod/* gRPC-Web requests to local Envoy (127.0.0.1:8080)
# - TLS via Let's Encrypt (managed by certbot)

server {
    server_name privatesecure.org www.privatesecure.org;

    root /var/www/privatesecure.org;
    index index.html;

    # --- gRPC-Web proxy to Envoy ---
    location /test/ {
        proxy_pass http://127.0.0.1:8080;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        # gRPC-Web uses large binary payloads
        client_max_body_size 4m;
        proxy_read_timeout 120s;
        proxy_send_timeout 120s;
    }

    location /prod/ {
        proxy_pass http://127.0.0.1:8080;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        client_max_body_size 4m;
        proxy_read_timeout 120s;
        proxy_send_timeout 120s;
    }

    # --- Static PWA (SPA fallback) ---
    location / {
        try_files $uri $uri/ /index.html;
    }

    # --- Cache static assets aggressively ---
    location ~* \.(js|css|wasm|svg|png|ico|webmanifest|woff2?)$ {
        expires 1y;
        add_header Cache-Control "public, immutable";
        try_files $uri =404;
    }

    # --- Security headers ---
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;

    listen [::]:443 ssl ipv6only=on;
    listen 443 ssl;
    ssl_certificate /etc/letsencrypt/live/privatesecure.org/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/privatesecure.org/privkey.pem;
    include /etc/letsencrypt/options-ssl-nginx.conf;
    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;
}

server {
    if ($host = www.privatesecure.org) {
        return 301 https://$host$request_uri;
    }
    if ($host = privatesecure.org) {
        return 301 https://$host$request_uri;
    }

    listen 80;
    listen [::]:80;
    server_name privatesecure.org www.privatesecure.org;
    return 404;
}

# --- DEQS proxy on port 7000 (TLS) ---
server {
    listen 7000;
    listen [::]:7000;
    server_name privatesecure.org www.privatesecure.org;

    # --- Proxy to DEQS deployment ---
    location / {
        proxy_pass https://ansite.tail9c57b.ts.net:7000;
        proxy_ssl_verify off;
        proxy_http_version 1.1;
        proxy_set_header Host ansite.tail9c57b.ts.net;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        client_max_body_size 4m;
        proxy_read_timeout 120s;
        proxy_send_timeout 120s;
    }
}
